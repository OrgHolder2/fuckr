        function createTerrain() {
            const geo = new THREE.PlaneGeometry(WORLD_SIZE, WORLD_SIZE, 64, 64);
            geo.rotateX(-Math.PI / 2);
            const pos = geo.attributes.position;
            for (let i = 0; i < pos.count; i++) {
                const x = pos.getX(i);
                const z = pos.getZ(i);
                const y = (Math.sin(x*0.04) + Math.cos(z*0.04)) * 3 + (Math.sin(x*0.1) * Math.cos(z*0.1)) * 2;
                pos.setY(i, y);
            }
            geo.computeVertexNormals();
            const mat = new THREE.MeshStandardMaterial({ color: 0x558833, flatShading: true });
            terrainMesh = new THREE.Mesh(geo, mat);
            terrainMesh.receiveShadow = true;
            scene.add(terrainMesh);

            // Trees
            for (let i = 0; i < 100; i++) {
                const x = (Math.random() - 0.5) * (WORLD_SIZE - 20);
                const z = (Math.random() - 0.5) * (WORLD_SIZE - 20);
                const y = getTerrainHeight(x, z);
                
                const tree = new THREE.Group();
                const trunk = new THREE.Mesh(new THREE.CylinderGeometry(0.4, 0.6, 2, 6), new THREE.MeshStandardMaterial({color: 0x654321}));
                trunk.position.y = 1;
                const leaves = new THREE.Mesh(new THREE.ConeGeometry(3, 5, 8), new THREE.MeshStandardMaterial({color: 0x228B22}));
                leaves.position.y = 3.5;
                tree.add(trunk);
                tree.add(leaves);
                tree.position.set(x, y, z);
                tree.traverse(o => { if(o.isMesh) { o.castShadow = true; o.receiveShadow = true; } });
                scene.add(tree);
            }
        }

        function getTerrainHeight(x, z) {
            return (Math.sin(x*0.04) + Math.cos(z*0.04)) * 3 + (Math.sin(x*0.1) * Math.cos(z*0.1)) * 2;
        }

        function spawnBots() {
            const limit = WORLD_SIZE/2 - 50;
            for (let i=0; i<BOT_COUNT; i++) {
                let x, z, distance;
                do {
                    x = (Math.random() - 0.5) * limit * 2;
                    z = (Math.random() - 0.5) * limit * 2;
                    distance = Math.sqrt(x*x + z*z);
                } while (distance < BOT_SPAWN_RADIUS); // Ensure bots are far away
                
                const bot = {
                    mesh: createHumanoidModel(0xe74c3c),
                    position: new THREE.Vector3(x, 100, z),
                    velocity: new THREE.Vector3(),
                    health: 100,
                    lastShot: 0,
                    speed: 4 + Math.random() * 3,
                    aggro: false, // Will chase if true
                    flashTime: 0 // Flash return fix
                };
                bot.mesh.userData.gun.visible = true;
                scene.add(bot.mesh);
                bots.push(bot);
            }
        }

        function createProjectile(start, dir, isPlayer) {
            const mesh = new THREE.Mesh(
                new THREE.SphereGeometry(0.1, 4, 4), 
                new THREE.MeshBasicMaterial({ color: isPlayer ? 0xffff00 : 0xff4400 })
            );
            mesh.position.copy(start);

            // Tail

            const tail = new THREE.Line(tailGeo, new THREE.LineBasicMaterial({ color: isPlayer ? 0xffff00 : 0xff4400, transparent: true, opacity: 0.5 }));
            mesh.add(tail);
            tail.lookAt(dir);

            projectiles.push({
                mesh: mesh,
                velocity: dir.clone().multiplyScalar(PROJECTILE_SPEED), // Faster bullets
                life: 1.5,
                fromPlayer: isPlayer,
                // New: store the previous position for raycasting
                lastPos: start.clone() 
            });
        }

        function checkProjectileCollision(p, step) {
            const start = p.lastPos;
            const end = p.mesh.position;
            const direction = step.clone().normalize();
            const distance = step.length();
            
            const raycaster = new THREE.Raycaster(start, direction, 0, distance);
            
            // Check for Player hit
            if (!p.fromPlayer) {
                const playerHitbox = new THREE.Box3().setFromObject(player.mesh);
                if (playerHitbox.containsPoint(end)) {
                    takeDamage(8);
                    return true;
                }
            }

            // Check for Bot hit (if it was a player projectile)
            if (p.fromPlayer) {
                for(let bot of bots) {
                    const botHitbox = new THREE.Box3().setFromObject(bot.mesh);
                    if (botHitbox.containsPoint(end)) {
                        applyBotDamage(bot, p.damage || 15);
                        return true;
                    }
                }
            }
            
            // Check for Terrain hit
            const intersects = raycaster.intersectObject(terrainMesh);
            if (intersects.length > 0) {
                return true;
            }

            return false;
        }

        function updateProjectiles(dt) {
            for (let i = projectiles.length - 1; i >= 0; i--) {
                const p = projectiles[i];
                p.life -= dt;
                
                p.lastPos.copy(p.mesh.position); // Store current pos
                const step = p.velocity.clone().multiplyScalar(dt);
                p.mesh.position.add(step);

                // Collision check
                if (checkProjectileCollision(p, step) || p.life <= 0) {
                    scene.remove(p.mesh);
                    projectiles.splice(i, 1);
                }
            }
        }

        function applyBotDamage(bot, dmg) {
             // Aggro Bot
            bot.aggro = true;

            bot.health -= dmg;
            showDamage(bot.mesh.position.clone().add(new THREE.Vector3(0, 1.5, 0)), dmg);
            
            // Flash Effect (FIXED: use a timer to reset)
            bot.flashTime = 0.05; // Flash for 50ms

            if (bot.health <= 0 && bots.includes(bot)) {
                scene.remove(bot.mesh);
                bots = bots.filter(b => b !== bot);
                gameState.enemiesAlive--;
                updateHUD();
                if (gameState.enemiesAlive === 0) showWin();
            }
        }

        function shootPlayer() {
            if (gameState.isReloading || !gameState.activeWeaponData) return;
            
            const weapon = gameState.activeWeaponData;
            const now = performance.now();
            if (now - lastShotTime < weapon.fireRate) return;

            if (gameState.ammo.clip <= 0) {
                reload();
                return;
            }

            lastShotTime = now;
            gameState.ammo.clip--;
            updateHUD();

            // Barrel Position
            const gunWorldPos = new THREE.Vector3();
            // Get position of the end of the barrel (gun's second child which is the barrel, offset by -0.4)
            player.mesh.userData.gun.children[1].getWorldPosition(gunWorldPos);

            // Center of Screen Ray
            const raycaster = new THREE.Raycaster();
            raycaster.setFromCamera(new THREE.Vector2(0, 0), camera); 
            
            const targetPoint = raycaster.ray.at(weapon.range, new THREE.Vector3());
            
            // Apply Shotgun Pellets or Single Shot
            const pellets = weapon.pellets || 1;
            
            for(let i=0; i<pellets; i++) {
                // Visual Tracer Direction
                const dir = new THREE.Vector3().subVectors(targetPoint, gunWorldPos).normalize();
                
                // Spread Logic
                const currentSpread = isADS ? weapon.spread * 0.3 : weapon.spread;
                
                dir.x += (Math.random()-0.5) * currentSpread;
                dir.y += (Math.random()-0.5) * currentSpread;
                dir.z += (Math.random()-0.5) * currentSpread;
                
                dir.normalize();

                const p = createProjectile(gunWorldPos, dir, true);
                p.damage = weapon.damage; // Attach damage to projectile for collision check
            }
        }

        function shootBot(bot) {
            const weapon = { damage: 8, fireRate: 800, range: 100 }; // Bot default gun
            
            if (performance.now() - bot.lastShot < weapon.fireRate) return;
            bot.lastShot = performance.now();
            
            // Barrel Position
            const gunPos = bot.mesh.position.clone().add(new THREE.Vector3(0, 1.2, 0.5));
            
            // Target direction
            const target = player.position.clone().add(new THREE.Vector3(0,1.5,0)); // Aim at head/chest
            let dir = new THREE.Vector3().subVectors(target, gunPos).normalize();
            
            // Miss Logic: Apply randomness based on distance and constant miss chance
            const dist = bot.position.distanceTo(player.position);
            const accuracyFactor = dist / 60; // Max spread at 60 units
            
            let spread = 0.05 * accuracyFactor;
            
            // Miss chance check
            if (Math.random() < BOT_MISS_CHANCE) {
                spread += 0.1; // Add extra spread for guaranteed miss
            }
            
            dir.x += (Math.random()-0.5) * spread;
            dir.y += (Math.random()-0.5) * spread;
            dir.z += (Math.random()-0.5) * spread;
            dir.normalize();

            const p = createProjectile(gunPos, dir, false);
            p.damage = weapon.damage;
            p.shooterPos = bot.position.clone(); // For directional damage indicator
        }

        function reload() {
            if (gameState.isReloading || gameState.ammo.clip === gameState.activeWeaponData?.clip || gameState.ammo.reserve === 0) return;
            gameState.isReloading = true;
            document.getElementById('reload-msg').style.display = 'block';
            setTimeout(() => {
                const maxClip = gameState.activeWeaponData.clip;
                const needed = maxClip - gameState.ammo.clip;
                const taken = Math.min(needed, gameState.ammo.reserve);
                gameState.ammo.clip += taken;
                gameState.ammo.reserve -= taken;
                gameState.isReloading = false;
                document.getElementById('reload-msg').style.display = 'none';
                updateHUD();
            }, 1500);
        }

        // --- Update Loop ---

        function updatePlayerLogic(dt) {
            if (!isGameActive) return;

            // Rotation
            player.mesh.rotation.y = cameraRotation.y;

            // Movement
            const dir = new THREE.Vector3();
            const fwd = new THREE.Vector3(0,0,-1).applyAxisAngle(new THREE.Vector3(0,1,0), cameraRotation.y);
            const right = new THREE.Vector3(1,0,0).applyAxisAngle(new THREE.Vector3(0,1,0), cameraRotation.y);

            if (keys.w) dir.add(fwd);
            if (keys.s) dir.sub(fwd);
            if (keys.d) dir.add(right);
            if (keys.a) dir.sub(right);
            if (dir.lengthSq() > 0) dir.normalize();

            player.position.add(dir.multiplyScalar(PLAYER_SPEED * dt));

            // Physics
            player.velocity.y -= GRAVITY * dt;
            player.position.y += player.velocity.y * dt;

            const groundY = getTerrainHeight(player.position.x, player.position.z);
            if (player.position.y < groundY) {
                player.position.y = groundY;
                player.velocity.y = 0;
                player.onGround = true;
            } else {
                player.onGround = false;
            }

            if (keys.space && player.onGround) {
                player.velocity.y = JUMP_FORCE;
            }

            const lim = WORLD_SIZE/2 - 2;
            player.position.x = THREE.MathUtils.clamp(player.position.x, -lim, lim);
            player.position.z = THREE.MathUtils.clamp(player.position.z, -lim, lim);

            player.mesh.position.copy(player.position);

            // Camera Follow
            const headPos = player.position.clone().add(new THREE.Vector3(0, 1.6, 0));
            
            const targetOffset = isADS ? camOffsetAds : CAM_OFFSET_NORMAL;
            const euler = new THREE.Euler(0, cameraRotation.y, 0, 'YXZ');
            const rotOffset = targetOffset.clone().applyEuler(euler);
            
            const camTarget = headPos.clone().add(rotOffset);
            camTarget.y += Math.sin(cameraRotation.x) * 2; 

            camera.position.lerp(camTarget, 0.2);
            
            const lookDir = new THREE.Vector3(0, 0, -1).applyEuler(new THREE.Euler(cameraRotation.x, cameraRotation.y, 0, 'YXZ'));
            camera.lookAt(headPos.clone().add(lookDir.multiplyScalar(10)));
            
            const isWalking = dir.lengthSq() > 0;
            animateModel(player.mesh, dt, isWalking, isADS, gameState.isReloading);
        }

        function checkBotVisibility(bot) {
            const dist = bot.position.distanceTo(player.position);
            if (dist > 100) return false; 

            const botEyePos = bot.position.clone().add(new THREE.Vector3(0, 1.6, 0));
            const dirToPlayer = new THREE.Vector3().subVectors(player.position, bot.position).normalize();
            
            const ray = new THREE.Raycaster(botEyePos, dirToPlayer);
            // Only check collision with terrain
            const intersects = ray.intersectObject(terrainMesh);
            
            const distToTerrain = (intersects.length > 0) ? intersects[0].distance : 9999;
            
            // Check if player is closer than the terrain hit
            if (dist < distToTerrain) {
                return true;
            }
            return false;
        }

        function updateBotsLogic(dt) {
            for(const bot of bots) {
                const dist = bot.position.distanceTo(player.position);
                
                // Handle flash
                if (bot.flashTime > 0) {
                    bot.flashTime -= dt;
                    bot.mesh.traverse(c => {
                        if(c.isMesh && c.material.emissive) {
                            c.material.emissive.setHex(0xffffff);
                        }
                    });
                } else {
                    // Turn off flash permanently
                    bot.mesh.traverse(c => {
                        if(c.isMesh && c.material.emissive) {
                            c.material.emissive.setHex(0x000000);
                        }
                    });
                }

                const canSee = checkBotVisibility(bot);
                let shouldChase = bot.aggro || (canSee && dist < 80);

                let moving = false;
                if (shouldChase && dist > 15) {
                    const dir = new THREE.Vector3().subVectors(player.position, bot.position).normalize();
                    dir.y = 0;
                    bot.position.add(dir.multiplyScalar(bot.speed * dt));
                    moving = true;
                }
                
                // Bot always looks at the player if aggro or can see
                if (shouldChase || canSee) {
                    bot.mesh.lookAt(player.position.x, bot.position.y, player.position.z);
                }

                // Shooting check: only if visible and close enough
                if (canSee && dist < 60) {
                    if (performance.now() - bot.lastShot > 1500) {
                        shootBot(bot);
                    }
                }


                const gy = getTerrainHeight(bot.position.x, bot.position.z);
                bot.position.y = gy;
                bot.mesh.position.copy(bot.position);

                animateModel(bot.mesh, dt, moving, false, false);
            }
        }

        function animateModel(mesh, dt, isMoving, ads, reloading) {
            const p = mesh.userData;
            mesh.userData.time = (mesh.userData.time || 0) + dt * 10;
            const t = mesh.userData.time;

            if (isMoving) {
                p.legL.rotation.x = Math.sin(t)*0.6;
                p.legR.rotation.x = Math.sin(t+Math.PI)*0.6;
                p.armL.rotation.x = Math.sin(t+Math.PI)*0.6;
                p.armR.rotation.x = Math.sin(t)*0.6;
            } else {
                p.legL.rotation.x = 0; p.legR.rotation.x = 0;
                p.armL.rotation.x = 0; p.armR.rotation.x = 0;
            }

            // Gun Arm Overrides
            if (p.gun.visible) {
                // The armR is already rotated 180 degrees (Math.PI) to face forward
                // Adjust X rotation for stance
                const armRotX = isMoving ? p.armR.rotation.x : 0;
                
                if (reloading) {
                    p.armR.rotation.x = -Math.PI/2 + 0.5;
                } else if (ads) {
                    p.armR.rotation.x = -Math.PI/2 - 0.2; 
                    p.armR.rotation.y = Math.PI - 0.2; // Add PI because of arm 180 flip
                } else {
                    p.armR.rotation.x = armRotX - 0.1; 
                    p.armR.rotation.y = Math.PI; // Standard 180 flipped
                }
            }
            
            if (p.pickaxe.visible && keys.mouse && !isMoving) {
                 p.armR.rotation.x = -Math.PI/2 + Math.sin(t*4) * 0.8;
                 p.armR.rotation.y = Math.PI;
            }
        }

        function checkInteractions() {
            let nearChest = false;
            for(const c of chests) {
                if(!c.opened && c.mesh.position.distanceTo(player.position) < 4) {
                    nearChest = true;
                    if(keys.e) {
                        c.opened = true;
                        c.mesh.material.color.setHex(0x555555);
                        
                        // Random Weapon Drop
                        const types = Object.keys(WEAPONS);
                        const rando = types[Math.floor(Math.random() * types.length)];
                        
                        const emptyIdx = gameState.inventory.indexOf(null);
                        if(emptyIdx !== -1) {
                            gameState.inventory[emptyIdx] = rando;
                            showWeaponMsg(`Picked up ${rando}!`);
                        } else {
                            if (gameState.activeSlot !== 0) {
                                gameState.inventory[gameState.activeSlot] = rando;
                                showWeaponMsg(`Swapped for ${rando}!`);
                            } else {
                                showWeaponMsg("Inventory Full! (Equip an empty slot first)");
                            }
                        }
                        
                        gameState.ammo.reserve += 40;
                        updateInventoryUI();
                        keys.e = false;
                    }
                }
            }
            document.getElementById('interaction-msg').style.display = nearChest ? 'block' : 'none';
        }

        function showDamage(pos, val) {
            const div = document.createElement('div');
            div.className = 'damage-popup';
            div.innerText = val;
            document.getElementById('ui-layer').appendChild(div);
            damagePopups.push({ el: div, pos: pos.clone(), life: 1.0 });
        }

        function showWeaponMsg(text) {
            const el = document.getElementById('weapon-msg');
            el.innerText = text;
            el.style.display = 'block';
            setTimeout(() => el.style.display = 'none', 2000);
        }

        function updateUI(dt) {
            for(let i=damagePopups.length-1; i>=0; i--) {
                const p = damagePopups[i];
                p.life -= dt;
                p.pos.y += dt; 
                
                const screen = p.pos.clone().project(camera);
                const x = (screen.x * 0.5 + 0.5) * window.innerWidth;
                const y = (-(screen.y * 0.5) + 0.5) * window.innerHeight;
                
                p.el.style.left = x + 'px';
                p.el.style.top = y + 'px';
                p.el.style.opacity = p.life;

                if(p.life <= 0) {
                    p.el.remove();
                    damagePopups.splice(i, 1);
                }
            }
        }

        function updateHUD() {
            document.getElementById('health-display').innerText = gameState.health;
            
            const item = gameState.inventory[gameState.activeSlot];
            if(WEAPONS[item]) {
                document.getElementById('ammo-display').innerText = `${gameState.ammo.clip} / ${gameState.ammo.reserve}`;
            } else {
                document.getElementById('ammo-display').innerText = "--";
            }
            
            document.getElementById('enemies-display').innerText = gameState.enemiesAlive;
            
            const ch = document.getElementById('crosshair');
            if (isADS) ch.className = 'ads';
            else if (gameState.activeWeaponData?.name === "Shotgun") ch.className = 'spread';
            else ch.className = '';
        }
        
        function showWin() { isGameActive = false; document.exitPointerLock(); document.getElementById('win-screen').style.display = 'flex'; }
        
        function takeDamage(n, sourcePos) {
             gameState.health -= n;
             updateHUD();
             
             // Radial Indicator Logic
             if (sourcePos) {
                const playerFwd = new THREE.Vector3(0, 0, -1).applyAxisAngle(new THREE.Vector3(0, 1, 0), cameraRotation.y);
                const dirToSource = new THREE.Vector3().subVectors(sourcePos, player.position).normalize();
                
                // Calculate angle (in radians) between player's forward vector and the direction to the source
                const angle = Math.atan2(playerFwd.z, playerFwd.x) - Math.atan2(dirToSource.z, dirToSource.x);
                let degrees = angle * (180 / Math.PI);

                // Normalize to 0-360
                degrees = (degrees + 360) % 360; 

                // Rotate the indicator so 0 degrees (up) is where the source is relative to player view
                // We want the arrow to point *away* from the shooter on the screen (i.e. where to look)
                
                // The angle we need to rotate the arrow is 180 degrees offset from the direction to source
                // degrees = degrees + 180; // This gives the direction *from* the source

                // Show the indicator and rotate the arrow
                damageIndicatorElement.style.display = 'block';
                damageIndicatorElement.arrow.style.opacity = 1.0;
                damageIndicatorElement.arrow.style.transform = `rotate(${-degrees}deg) translateY(-50%)`;

                clearTimeout(damageIndicatorElement.timer);
                damageIndicatorElement.timer = setTimeout(() => {
                    damageIndicatorElement.arrow.style.opacity = 0;
                }, 500);
             }


             if(gameState.health <= 0) { 
                isGameActive = false; 
                document.exitPointerLock(); 
                document.getElementById('death-screen').style.display = 'flex'; 
            }
        }

        // --- Input ---
        function onKeyDown(e) {
            const k = e.key.toLowerCase();
            if(keys[k] !== undefined) keys[k] = true;
            if(k === 'e') keys.e = true;
            if(['1','2','3','4','5'].includes(k)) {
                gameState.activeSlot = parseInt(k) - 1;
                updateInventoryUI();
            }
        }
        function onKeyUp(e) {
            const k = e.key.toLowerCase();
            if(keys[k] !== undefined) keys[k] = false;
            if(k === 'e') keys.e = false;
        }
        function onMouseDown(e) {
            if(!isGameActive) return;
            if(e.button === 0) {
                keys.mouse = true;
                if(gameState.activeWeaponData || gameState.inventory[gameState.activeSlot] === "Pickaxe") shootPlayer();
            }
            if(e.button === 2) isADS = true;
        }
        function onMouseUp(e) {
            if(e.button === 0) keys.mouse = false;
            if(e.button === 2) isADS = false;
        }
        function onMouseMove(e) {
            if(document.pointerLockElement === document.body && isGameActive) {
                const sens = isADS ? 0.001 : 0.002;
                cameraRotation.y -= e.movementX * sens;
                cameraRotation.x -= e.movementY * sens;
                cameraRotation.x = Math.max(-1.4, Math.min(1.4, cameraRotation.x));
            }
        }
        function onResize() {
            camera.aspect = window.innerWidth/window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        }

        function animate() {
            requestAnimationFrame(animate);
            const time = performance.now();
            const dt = Math.min((time - lastTime)/1000, 0.1);
            lastTime = time;

            if (isGameActive) {
                updatePlayerLogic(dt);
                updateBotsLogic(dt);
                updateProjectiles(dt);
                checkInteractions();
                updateUI(dt);
            }

            renderer.render(scene, camera);
        }

        window.onload = function() {
            init();
        }
